class LogCapture:
    """Redirige les print() vers un widget Text Tkinter ET vers la console Python."""

    def __init__(self, text_widget):
        self.text_widget = text_widget
        self.original_stdout = sys.__stdout__  # Console Python originale
        self.original_stderr = sys.__stderr__

    def write(self, message):
        """Écrit dans le widget Text ET dans la console Python."""
        # ✅ TOUJOURS afficher dans la console Python originale
        self.original_stdout.write(message)
        self.original_stdout.flush()

        if message.strip():  # Ignorer les lignes vides
            log_line = message.strip() + "\n"

            def _update_widget():
                self.text_widget.insert("end", log_line)

                lines = self.text_widget.get("1.0", "end").split('\n')
                if len(lines) > 100:
                    self.text_widget.delete("1.0", "2.0")

                self.text_widget.see("end")
                self.text_widget.update_idletasks()

            self.text_widget.after(0, _update_widget)


    def flush(self):
        """Nécessaire pour être compatible avec sys.stdout."""
        self.original_stdout.flush()

def monitor_directory(directory, callback, shared_state):
    processed_files = set()
    start_time = None

    print("🔍 Surveillance du dossier démarrée...")

    while not shared_state["stop"]:
        try:
            all_files = sorted([
                os.path.join(directory, f) for f in os.listdir(directory)
                if f.endswith(".txt")
            ], key=lambda x: os.path.getmtime(x))
        except (FileNotFoundError, OSError) as e:
            print(f"⚠️ Erreur lecture du dossier : {e}")
            time.sleep(2)
            shared_state["monitor_running"] = False
            print("✅ Surveillance arrêtée.")
            continue


        new_files = [f for f in all_files if f not in processed_files]

        if new_files and start_time is None:
            start_time = os.path.getmtime(new_files[0])

        for file_path in new_files:
            if shared_state["stop"]:
                print("⏸️ Stop request : interruption of file processing.")
                break
            try:
                elapsed_time = os.path.getmtime(file_path) - start_time
                print(f"▶️ Processing of : {file_path}")
                callback(file_path, elapsed_time)
            except Exception as e:
                import traceback
                print(f"💥 Error during process of {file_path} : {e}")
                traceback.print_exc()
            processed_files.add(file_path)

        time.sleep(2)

def apply_savgol_filter(data, window_length, polyorder):
    """
    Applique Savitzky–Golay en ajustant automatiquement (window_length, polyorder)
    aux contraintes du signal (taille, parité, ordre).
    """
    y = np.asarray(data, dtype=float)
    wl, po = adjust_savgol_params(len(y), window_length, polyorder)
    return savgol_filter(y, window_length=wl, polyorder=po)

def adjust_savgol_params(n: int, wl: int, po: int):
    """
    Ajuste (wl, po) pour Savitzky–Golay:
    - wl >= 3 et impair
    - wl <= n-1 (ou n-2 si nécessaire) pour rester valide
    - wl > po (sinon wl = po + 1, puis forcer impair)
    Retourne (wl, po) ajustés.
    """
    wl = int(max(3, wl))
    po = int(po)

    # wl ne doit pas dépasser la taille utile
    if wl > n:
        wl = n - 1 if (n - 1) % 2 == 1 else n - 2
    wl = max(3, wl)

    # wl doit être impair
    if wl % 2 == 0:
        wl -= 1

    # wl doit être strictement > po
    if wl <= po:
        wl = po + 1
        if wl % 2 == 0:
            wl += 1
        if wl > n:
            wl = n - 1 if (n - 1) % 2 == 1 else n - 2
            wl = max(3, wl)

    return wl, po


def calculate_baseline_arpls(data, potential, potential_range, lam=LAM,
                             weight_extremes=WEIGHT_EXTREMES,
                             weight_central=WEIGHT_CENTRAL,
                             extremes_percentage=EXTREMES_PERCENTAGE,
                             max_iter=MAX_ITER, tol=TOL,
                             filename=None):
    # Utiliser potential_range dynamique ici !
    potential_indices = (potential >= potential_range[0]) & (potential <= potential_range[1])

    """
    Calcule la baseline en utilisant la méthode ARPLS avec des paramètres dynamiques.
    """
    if filename is None:
        print("⚠️ Le paramètre 'filename' est None ! Vérifiez l'appel de la fonction.")


    # Filtrer la plage de potentiels
    data_filtered = data[potential_indices]
    potential_filtered = potential[potential_indices]

    if len(data_filtered) < 10:
        print(f"⚠️ Pas assez de données dans la plage de potentiel pour {filename}")
        return None, potential_filtered

    def calculate_adaptive_weights(data_length, data, potential,
                                   weight_extremes=1e10,
                                   weight_central=50,
                                   peak_zone=(-0.32, -0.24),
                                   anchor_zones=[(-0.40, -0.36), (-0.15, -0.10)]):
        weight = np.ones(data_length) * weight_central
        potential = np.array(potential)

        # Forcer les extrémités à coller
        for zone in anchor_zones:
            mask = (potential >= zone[0]) & (potential <= zone[1])
            weight[mask] = weight_extremes

        # Ignorer la zone du pic
        mask_peak = (potential >= peak_zone[0]) & (potential <= peak_zone[1])
        weight[mask_peak] = 0.001

        return weight

    weight = calculate_adaptive_weights(len(data_filtered), data_filtered, potential_filtered)

    try:
        # Calcul de la baseline avec ARPLS
        baseline, _ = arpls(data_filtered, lam=lam, diff_order=2, weights=weight, max_iter=max_iter, tol=tol)
        return baseline, potential_filtered
    except Exception as e:
        print(f"⚠️ Error in ARPLS for {filename}: {e}")
        return None, potential_filtered

def validate_baseline(data, baseline, percentage=0.1, max_offset=0.0015, max_bias=0.2, max_slope_diff=0.5):
    """
    Valide la qualité de la baseline aux extrémités en vérifiant :
    - l'écart moyen
    - le biais
    - la pente (tendance) par rapport au signal

    Args:
    - data : signal de courant (array)
    - baseline : baseline estimée (array)
    - percentage : portion du signal à analyser sur les extrémités
    - max_offset : tolérance d'écart moyen
    - max_bias : tolérance de biais moyen
    - max_slope_diff : tolérance sur la différence de pente

    Returns:
    - bool : True si la baseline est jugée correcte, False sinon
    """
    n = len(data)
    num = max(3, int(n * percentage))  # au moins 3 points pour une pente

    def check_side(data_side, base_side):
        diff = data_side - base_side
        offset = np.mean(np.abs(diff))
        bias = np.mean(diff)

        # Approximation de la pente (moindre carrés)
        x = np.arange(len(data_side))
        slope_data = np.polyfit(x, data_side, 1)[0]
        slope_base = np.polyfit(x, base_side, 1)[0]
        slope_diff = abs(slope_data - slope_base)

        return offset, bias, slope_diff

    left_offset, left_bias, left_slope = check_side(data[:num], baseline[:num])
    right_offset, right_bias, right_slope = check_side(data[-num:], baseline[-num:])

    conditions = [
        left_offset <= max_offset,
        right_offset <= max_offset,
        abs(left_bias) <= max_bias,
        abs(right_bias) <= max_bias,
        left_slope <= max_slope_diff,
        right_slope <= max_slope_diff
    ]

    if not all(conditions):
        return False

    return True

# Calcul de la hauteur du pic en soustrayant la baseline
def calculate_peak_height(data, baseline, peak_index):
    data_corrected = data - baseline  # Soustraction de la baseline
    peak_height = data_corrected.iloc[peak_index]  # Hauteur du pic après correction
    return peak_height

        df = pd.read_csv(

            skiprows=nb_header + 1,
            names=["Potentiel (V)", "Courant (uA)"],
            encoding='latin1',
            decimal=decimal_sep
        )

    else:
        # Lecture brute pour détecter le séparateur décimal
        with open(file_path, 'r', encoding='latin1') as f:
            lines = f.readlines()

        data_line = lines[1].strip()
        decimal_sep = ',' if ',' in data_line else '.'

        # Lecture avec noms de colonnes imposés
        df = pd.read_csv(
            file_path,
            sep='\t',

            skiprows=1,  # saute la ligne d'en-tête

        )

        # Conversion du courant en µA
        df["Courant (uA)"] = df["Courant (A)"].astype(float) * 1e6
        df = df.drop(columns=["Courant (A)"])

    return df

def hampel_filter_1d(y, window_size=HAMPEL_WINDOW, n_sigma=HAMPEL_SIGMA, max_passes=HAMPEL_MAX_PASS, use_derivative=True):
    """
    Filtre de Hampel renforcé:
    - Multi-passe (max_passes)
    - Détection sur le signal et (optionnel) sur sa dérivée
    - Remplacement par médiane locale ou interpolation linéaire robuste

    Retourne (y_repaired, spike_mask_final)
    """

    y = np.asarray(y, dtype=float)
    n = len(y)
    w = int(window_size)
    if w < 3:
        w = 3
    if w % 2 == 0:
        w += 1

    def _rolling_median(arr, win):
        return pd.Series(arr).rolling(win, center=True, min_periods=3).median().to_numpy()

    def _mad(arr, center):
        return pd.Series(np.abs(arr - center)).rolling(w, center=True, min_periods=3).median().to_numpy()

    def _hampel_pass(y_in):
        med = _rolling_median(y_in, w)
        sigma = 1.4826 * _mad(y_in, med)
        diff = np.abs(y_in - med)
        mask = (sigma > 0) & (diff > n_sigma * sigma)
        return mask, med

    def _derivative(arr):
        # dérivée centrale simple; bords en avant/arrière
        d = np.empty_like(arr)
        d[1:-1] = 0.5 * (arr[2:] - arr[:-2])
        d[0] = arr[1] - arr[0]
        d[-1] = arr[-1] - arr[-2]
        return d

    def _replace_value(y_arr, idx, med_local, spike_mask):
        # Si la médiane locale est finie, l’utiliser
        if np.isfinite(med_local[idx]):
            return med_local[idx]
        # sinon: interpolation linéaire robuste avec voisins non-spikes
        k = max(3, w // 2)
        start = max(0, idx - k)
        end   = min(n, idx + k + 1)
        # indices non spikes dans la fenêtre
        neighbors = [j for j in range(start, end) if j != idx and not spike_mask[j] and np.isfinite(y_arr[j])]
        if len(neighbors) >= 2:
            # fit linéaire local (x=j, y=y[j])
            x = np.array(neighbors, dtype=float)
            yy = y_arr[neighbors]
            try:
                coeff = np.polyfit(x, yy, 1)
                return float(np.polyval(coeff, idx))
            except Exception:
                return float(np.median(yy))
        elif len(neighbors) == 1:
            return float(y_arr[neighbors[0]])
        else:
            return y_arr[idx]  # fallback: garder tel quel

    y_work = y.copy()
    spike_mask_final = np.zeros_like(y, dtype=bool)

    for p in range(max_passes):
        # Hampel sur le signal
        mask_sig, med_sig = _hampel_pass(y_work)

        # Hampel sur la dérivée
        if use_derivative:
            dy = _derivative(y_work)
            mask_der, med_der = _hampel_pass(dy)
            # projeter les spikes de dérivée sur le signal (mêmes indices)
            mask = mask_sig | mask_der
        else:
            mask = mask_sig

        # Nouveaux indices à corriger (éviter de re-traiter ceux déjà remplacés)
        new_spikes = mask & (~spike_mask_final)

        if not np.any(new_spikes):
            break  # plus rien à corriger

        # Remplacement
        y_repaired = y_work.copy()
        idxs = np.where(new_spikes)[0]
        for idx in idxs:
            y_repaired[idx] = _replace_value(y_work, idx, med_sig, mask)

        y_work = y_repaired
        spike_mask_final = spike_mask_final | new_spikes

    return y_work, spike_mask_final

def compute_peak_height_from_file(file_path, params):
    """
    Calcule la hauteur de pic (µA) d'un fichier SWV en réutilisant le pipeline
    principal (compute_file_result), pour éviter toute divergence entre les deux.
    Retourne (peak_height, peak_potential, pot, y_filt, baseline_arr, peak_index).
    """
    payload = compute_file_result(
        file_path,
        potential_range=params["plage_potentiel"],
        peak_range=params["position_pic"],
        lam=params["lambda"],
        method=params.get("baseline_method", "ARPLS"),
        polyorder=params["polyorder"],
        window_length=params["window_length"],
    )

    if payload.get("type") != "full":
        raise RuntimeError(payload.get("message", "Erreur inconnue lors du traitement du fichier de référence."))

    df = payload["df"]
    pot = df["Potentiel (V)"].to_numpy(dtype=float)
    y_filt = df["courant_filtre"].to_numpy(dtype=float)
    baseline_arr = payload["baseline_full"]
    peak_index = payload["peak_index"]
    peak_potential = float(pot[peak_index])
    peak_height = float(payload["row"]["Peak height (µA)"])

    ax_disp = shared_state.get("ax")
    canvas_disp = shared_state.get("canvas")
    if ax_disp is not None:
        update_fixed_peak_display(
            ax_disp,
            canvas_disp,
            file_name=os.path.basename(file_path),
            peak_height_uA=peak_height,
            peak_V=peak_potential,
            visible=True
        )

    return peak_height, peak_potential, pot, y_filt, baseline_arr, peak_index

def find_peak_robust(corrected_signal, potential, peak_range, potential_range,
                     spike_mask=None, min_width_pts=3, prominence_frac=0.05):
    """
    Trouve un pic robuste dans le signal corrigé (courant_filtre - baseline), en évitant
    les spikes isolés et en imposant une largeur minimale.
    Retourne l’indice global du pic choisi.
    """
    from scipy.signal import find_peaks, medfilt
    from numpy.polynomial import Polynomial

    y = np.asarray(corrected_signal, dtype=float)
    pot = np.asarray(potential, dtype=float)

    mask_pot = (pot >= potential_range[0]) & (pot <= potential_range[1])
    pot_p = pot[mask_pot]
    y_p = y[mask_pot]
    idx_global_p = np.nonzero(mask_pot)[0]

    mask_peak = (pot_p >= peak_range[0]) & (pot_p <= peak_range[1])
    pot_w = pot_p[mask_peak]
    y_w = y_p[mask_peak]
    idx_global_w = idx_global_p[np.nonzero(mask_peak)[0]]

    if y_w.size == 0:
        return int(idx_global_p[int(np.nanargmax(np.abs(y_p)))]) if idx_global_p.size else 0

    # Ignorer les indices "spikes" si fournis
    if spike_mask is not None and spike_mask.size == y.size:
        spike_local = spike_mask[idx_global_w]
    else:
        spike_local = np.zeros_like(y_w, dtype=bool)

    ymax = np.nanmax(np.abs(y_w)) if np.isfinite(np.nanmax(np.abs(y_w))) else 0.0
    prom = max(1e-12, prominence_frac * ymax)
    try:
        peaks, props = find_peaks(y_w, prominence=prom, distance=2, width=min_width_pts)
        keep = [i for i in range(len(peaks)) if not spike_local[peaks[i]]]
        peaks = peaks[keep]
        if len(peaks) > 0:
            best_local = peaks[int(np.nanargmax(y_w[peaks]))]
            return int(idx_global_w[best_local])
    except Exception:
        pass

    # Fallback: polynomial sur la fenêtre
    try:
        deg = min(5, max(2, len(y_w) - 1))
        coeffs = np.polyfit(pot_w, y_w, deg)
        p = Polynomial(coeffs[::-1])
        roots = p.deriv(1).roots()
        r_real = roots[np.isreal(roots)].real
        r_real = r_real[(r_real >= peak_range[0]) & (r_real <= peak_range[1])]
        if r_real.size > 0:
            vals = [np.interp(r, pot_w, y_w) for r in r_real]
            chosen = float(r_real[int(np.nanargmax(vals))])
            loc = int(np.nanargmin(np.abs(pot_w - chosen)))
            if not spike_local[loc]:
                return int(idx_global_w[loc])
    except Exception:
        pass

    # Fallback final: median-filter et argmax hors spikes
    try:
        y_m = medfilt(y_w, kernel_size=5 if len(y_w) >= 5 else 3)
        candidates = np.where(~spike_local)[0]
        if candidates.size > 0:
            best = candidates[int(np.nanargmax(y_m[candidates]))]
            return int(idx_global_w[best])
        else:
            est = float(np.nanpercentile(y_w, 95))
            best = int(np.nanargmin(np.abs(y_w - est)))
            return int(idx_global_w[best])
    except Exception:
        best = int(np.nanargmax(y_w))
        return int(idx_global_w[best])

def compute_file_result(file_path, elapsed_time=None, potential_range=None, peak_range=None,
                         lam=None, method="ARPLS", polyorder=1, apply_baseline=True,
                         window_length=11, pos_x=None, pos_y=None):
    """Calcul PUR : lecture, filtrage, baseline, pic. Ne touche à AUCUN widget Tkinter/matplotlib.
    Peut donc être appelée en toute sécurité depuis un thread de fond."""
    if elapsed_time is None:
        elapsed_time = 0
    if potential_range is None:
        potential_range = POTENTIAL_RANGE
    if peak_range is None:
        peak_range = PEAK_RANGE
    if lam is None:
        lam = LAM

    print(f"📂 Processing : {os.path.basename(file_path)}")

    try:
        df = lire_donnees(file_path)
        y_series = (df['Courant (uA)'] if 'Courant (uA)' in df.columns else df.get('Courant inverse (uA)', df.iloc[:, 1])).astype(float)
        y_raw = y_series.to_numpy()
        pot = df['Potentiel (V)'].to_numpy()

        # Hampel
        try:
            y_hampel, spike_mask = hampel_filter_1d(y_raw, window_size=11, n_sigma=2.5, max_passes=3, use_derivative=True)
            y_raw = y_hampel
            repaired_auto = bool(np.any(spike_mask))
        except Exception as e:
            print(f"⚠️ Hampel strict indisponible, ignore: {e}")
            spike_mask = np.zeros_like(y_raw, dtype=bool)
            repaired_auto = False

        # Auto-inversion
        mask_pot = (pot >= potential_range[0]) & (pot <= potential_range[1])
        mask_peak = (pot >= peak_range[0]) & (pot <= peak_range[1])
        mask_both = mask_pot & mask_peak
        try:
            y_window = y_raw[mask_both] if mask_both.any() else (y_raw[mask_pot] if mask_pot.any() else y_raw)
            invert = (y_window.size > 0 and y_window[np.argmax(np.abs(y_window))] < 0)
        except Exception as e:
            print(f"⚠️ Auto-inversion ignorée (raison: {e})")
            invert = False

        y_used = (-y_raw if invert else y_raw)
        df['courant_filtre'] = apply_savgol_filter(y_used, window_length=window_length, polyorder=polyorder)

        mask_plot = (df['Potentiel (V)'] >= potential_range[0]) & (df['Potentiel (V)'] <= potential_range[1])
        df_plot = df[mask_plot] if not df[mask_plot].empty else df

        if not apply_baseline:
            return {"type": "no_baseline", "file_path": file_path, "df_plot": df_plot}

        # Baseline (avec tri systématique)
        if method == "ARPLS":
            baseline, filtered_potential = calculate_baseline_arpls(
                df['courant_filtre'], df['Potentiel (V)'], potential_range, lam=lam,
                filename=os.path.basename(file_path)
            )
        else:
            baseline, filtered_potential = calculate_baseline_aSpls(
                df['courant_filtre'], df['Potentiel (V)'], potential_range, lam=lam,
                filename=os.path.basename(file_path)
            )

        order = np.argsort(np.asarray(filtered_potential))
        interp_baseline = interp1d(
            np.asarray(filtered_potential)[order], np.asarray(baseline)[order],
            kind='linear', bounds_error=False, fill_value="extrapolate"
        )
        baseline_interpolated = pd.Series(interp_baseline(df['Potentiel (V)'].values), index=df.index)

        is_valid = validate_baseline(df['courant_filtre'], baseline_interpolated)
        method_used = method
        if not is_valid:
            print("⛔ Baseline invalide. Recalcul avec méthode alternative.")
            if method == "ARPLS":
                baseline, filtered_potential = calculate_baseline_aSpls(
                    df['courant_filtre'], df['Potentiel (V)'], potential_range, lam=lam,
                    filename=os.path.basename(file_path)
                )
                method_used = "ASPLS"
            else:
                baseline, filtered_potential = calculate_baseline_arpls(
                    df['courant_filtre'], df['Potentiel (V)'], potential_range, lam=lam,
                    filename=os.path.basename(file_path)
                )
                method_used = "ARPLS"
            order = np.argsort(np.asarray(filtered_potential))
            interp_baseline = interp1d(
                np.asarray(filtered_potential)[order], np.asarray(baseline)[order],
                kind='linear', bounds_error=False, fill_value="extrapolate"
            )
            baseline_interpolated = pd.Series(interp_baseline(df['Potentiel (V)'].values), index=df.index)
            is_valid = validate_baseline(df['courant_filtre'], baseline_interpolated)

        corrected = df['courant_filtre'] - baseline_interpolated
        peak_index = find_peak_robust(
            corrected.values, df['Potentiel (V)'].values, peak_range, potential_range,
            spike_mask=spike_mask, min_width_pts=3, prominence_frac=0.05
        )
        peak_potential = df['Potentiel (V)'].iloc[peak_index]
        peak_height = float(corrected.iloc[peak_index])

        shared_state["single_trace_peak"] = {
            "height_uA": peak_height, "peak_V": float(peak_potential),
            "file": os.path.basename(file_path)
        }
        print(f"📍 Peak detected at {peak_potential:.3f} V | Corrected height : {peak_height:.5f} µA")

        if shared_state.get("single_trace_mode", False):
            peak_var_text = f"Pic: {peak_height:.4f} µA  @ {float(peak_potential):.3f} V"
        else:
            peak_var_text = "Pic: —"

        baseline_plot = baseline_interpolated[mask_plot] if mask_plot.any() else baseline_interpolated

        # Positions (logique inchangée)
        if pos_x is None or pos_y is None:
            pos_map = shared_state.get("pos_by_file", {})
            base = os.path.basename(file_path).lower()
            if base in pos_map:
                pos_x, pos_y = pos_map[base]
            elif secm_positions:
                m = re.search(r"SWV_\((\d+)\)", os.path.basename(file_path))
                if m:
                    idx = int(m.group(1))
                    if 0 <= idx < len(secm_positions):
                        pos_x = secm_positions[idx].get("X_rel", secm_positions[idx].get("X rel. [µm]"))
                        pos_y = secm_positions[idx].get("Y_rel", secm_positions[idx].get("Y rel. [µm]"))
            if (pos_x is None or pos_y is None) and secm_positions and len(results) == 0:
                pos_x = secm_positions[0].get("X_rel", secm_positions[0].get("X rel. [µm]"))
                pos_y = secm_positions[0].get("Y_rel", secm_positions[0].get("Y rel. [µm]"))

        row = {
            "Nom du fichier": os.path.basename(file_path),
            "Cumulated time (s)": int(elapsed_time),
            "Position du pic (V)": peak_potential,
            "Peak height (µA)": peak_height,
            "X rel. (µm)": pos_x,
            "Y rel. (µm)": pos_y,
            "Courant normalisé": None,
            "Signal change (%)": None,
            "Baseline correcte": is_valid,
            "Méthode baseline": method_used,
            "Auto réparé": repaired_auto
        }

        return {
            "type": "full", "file_path": file_path, "df": df, "df_plot": df_plot,
            "baseline_plot": baseline_plot, "baseline_full": np.asarray(baseline_interpolated, dtype=float),
            "peak_index": peak_index,
            "peak_var_text": peak_var_text, "row": row
        }

    except Exception as e:
        print(f"💥 Error during process : {e}")
        return {"type": "error", "file_path": file_path, "message": str(e)}


def render_file_result(ax, canvas, payload):
    """Affichage PUR : tout ce qui touche ax/canvas/status_var/results.
    Doit TOUJOURS être exécutée depuis le thread principal Tkinter."""
    file_path = payload["file_path"]

    if payload["type"] == "error":
        if status_var is not None:
            status_var.set(f"❌ Error during process of : {os.path.basename(file_path)}")
        return

    if payload["type"] == "no_baseline":
        df_plot = payload["df_plot"]
        ax.plot(df_plot['Potentiel (V)'], df_plot['courant_filtre'], label=os.path.basename(file_path),
                linewidth=1.2, zorder=3)
        ax.set_xlabel("Potential (V)")
        ax.set_ylabel("Current (µA)")
        ax.set_title("Voltammograms")
        ax.relim()
        ax.autoscale_view()
        canvas.draw()
        if status_var is not None:
            status_var.set(f"✅ Fichier tracé (sans baseline) : {os.path.basename(file_path)}")
        return

    # type == "full"
    df, df_plot = payload["df"], payload["df_plot"]
    baseline_plot, peak_index = payload["baseline_plot"], payload["peak_index"]

    ax.plot(df_plot['Potentiel (V)'], df_plot['courant_filtre'], label='_nolegend_')
    ax.plot(df_plot['Potentiel (V)'], baseline_plot, '--', color='red', linewidth=1.5, zorder=4, label='_nolegend_')
    ax.scatter(df['Potentiel (V)'].iloc[peak_index], df['courant_filtre'].iloc[peak_index], color='purple', s=60, zorder=5)
    ax.set_xlabel("Potential (V)")
    ax.set_ylabel("Current (µA)")
    canvas.draw_idle()

    update_peak_var_display(ax, canvas, payload["peak_var_text"])

    results.append(payload["row"])
    # ✅ Recalculer la normalisation une seule fois avec tous les résultats
    recalculate_normalization()

    if status_var is not None:
        status_var.set(f"✅ Processed file: {os.path.basename(file_path)}")


def process_single_file(file_path, ax, canvas, elapsed_time=None, potential_range=None, peak_range=None,
                        lam=None, method="ARPLS", polyorder=1, apply_baseline=True, window_length=11,
                        pos_x=None, pos_y=None):
    """Version synchrone inchangée (calcul + affichage à la suite) — utilisée uniquement
    depuis le thread principal (ex: 'Select file')."""
    payload = compute_file_result(file_path, elapsed_time, potential_range, peak_range, lam,
                                   method, polyorder, apply_baseline, window_length, pos_x, pos_y)
    render_file_result(ax, canvas, payload)


def process_file_threadsafe(file_path, ax, canvas, elapsed_time=None, potential_range=None, peak_range=None,
                             lam=None, method="ARPLS", polyorder=1, apply_baseline=True, window_length=11,
                             pos_x=None, pos_y=None):
    """Version pour le thread de surveillance : calcule ici (thread de fond),
    mais délègue tout l'affichage au thread principal via la queue, et ATTEND
    que ce soit fait avant de continuer (préserve l'ordre exact d'avant)."""
    payload = compute_file_result(file_path, elapsed_time, potential_range, peak_range, lam,
                                   method, polyorder, apply_baseline, window_length, pos_x, pos_y)
    gui_queue.put((ax, canvas, payload))
    gui_queue.join()


def load_secm_positions(file_path=None):
    """
    Charge un fichier .dat de positions SECM (Sensolytics).
    Stocke les coordonnées X, Y et le nom de fichier associé.
    """
    global secm_positions
    if file_path is None:
        file_path = filedialog.askopenfilename(
            filetypes=[("Fichiers SECM .dat", "*.dat"), ("Tous fichiers", "*.*")]
        )
    if not file_path:
        print("❌ Aucun fichier de positions sélectionné.")
        return

    try:
        with open(file_path, "r", encoding="latin1") as f:
            lines = f.readlines()

        header_idx = None
        for i, line in enumerate(lines):
            ls = line.lstrip()
            if ls.startswith("#") and (("X rel." in ls and "Y rel." in ls) or ("X [µm]" in ls and "Y [µm]" in ls)):
                header_idx = i
                break

        if header_idx is None:
            print("⚠️ En-tête introuvable dans le fichier .dat")
            return

        from io import StringIO
        lines[header_idx] = lines[header_idx].lstrip("#").lstrip()
        if not lines[header_idx].endswith("\n"):
            lines[header_idx] += "\n"

        data_str = "".join(lines[header_idx:])

        df = pd.read_csv(
            StringIO(data_str),
            sep=",",
            engine="python",
            skipinitialspace=True,
            quotechar='"',
            escapechar='\\',
            comment='#',
            on_bad_lines='skip',
            skip_blank_lines=True
        )

        df.columns = [col.strip() for col in df.columns]
        df.columns = df.columns.str.replace("Âµ", "µ", regex=False)

        def _canon(s: str) -> str:
            s = s.replace('\xa0', ' ')
            s = s.strip().lower()
            s = s.replace('µ', 'u')
            s = re.sub(r'[\[\]\(\)\.,"“”]', '', s)
            s = re.sub(r'\s+', ' ', s)
            return s

        canon_map = {c: _canon(c) for c in df.columns}
        x_rel_col = next((c for c, cn in canon_map.items() if 'x' in cn and 'rel' in cn), None)
        y_rel_col = next((c for c, cn in canon_map.items() if 'y' in cn and 'rel' in cn), None)

        if x_rel_col is None or y_rel_col is None:
            print(f"⚠️ Colonnes X rel. et Y rel. manquantes. Colonnes lues: {list(df.columns)}")
            return

        last_col = df.columns[-1]

        df["_x_rel_num"] = pd.to_numeric(df[x_rel_col], errors='coerce')
        df["_y_rel_num"] = pd.to_numeric(df[y_rel_col], errors='coerce')
        df = df[df["_x_rel_num"].notna() & df["_y_rel_num"].notna()].drop(columns=["_x_rel_num", "_y_rel_num"])

        df.rename(columns={x_rel_col: "X_rel", y_rel_col: "Y_rel", last_col: "File"}, inplace=True)

        # Normaliser la colonne File -> basename en minuscules (clé de jointure robuste)
        df["File"] = df["File"].astype(str).str.strip().apply(lambda x: os.path.basename(x)).str.lower()

        secm_positions = df[["X_rel", "Y_rel", "File"]].to_dict(orient="records")
        shared_state["df_secm_positions"] = df[["X_rel", "Y_rel", "File"]].copy()
        # Mapping direct pour accès rapide
        shared_state["pos_by_file"] = {row["File"]: (row["X_rel"], row["Y_rel"]) for _, row in df.iterrows()}

        print(f"✅ {len(secm_positions)} SECM positions loaded")
        if secm_positions:
            print(f"   Exemple: {secm_positions[0]}")

    except Exception as e:
        print(f"💥 Erreur lors du chargement du fichier SECM : {e}")



def run_secm_mapping_dat():
    global secm_positions

    if not secm_positions:
        print("⚠️ Aucune position SECM chargée, charge d’abord le fichier .dat")
        return

    file_paths = filedialog.askopenfilenames(
        title="Sélectionne les fichiers SWV à associer aux positions (.dat)",
        filetypes=[("Text files", "*.txt"), ("All files", "*.*")]
    )
    if not file_paths:
        print("🚫 Aucun fichier sélectionné")
        return

    sorted_files = sorted(file_paths, key=lambda f: os.path.getmtime(f))
    try:
        selected_dir = os.path.commonpath([os.path.dirname(p) for p in sorted_files])
    except ValueError:
        selected_dir = os.path.dirname(sorted_files[0])
    shared_state["directory"] = selected_dir
    print(f"📁 Export file : {selected_dir}")

    ax = shared_state.get("ax")
    canvas = shared_state.get("canvas")
    if ax is None or canvas is None:
        print("⚠️ L'UI n'est pas initialisée (ax/canvas manquants). Lance d'abord la fenêtre principale.")
        return

    params = get_current_parameters()
    if params is None:
        print("⚠️ Parameters not valide.")
        return

    file_times = [os.path.getmtime(f) for f in sorted_files]
    start_time = file_times[0]

    for idx, file_path in enumerate(sorted_files):
        if idx < len(secm_positions):
            pos_x = secm_positions[idx].get("X_rel", secm_positions[idx].get("X rel. [µm]"))
            pos_y = secm_positions[idx].get("Y_rel", secm_positions[idx].get("Y rel. [µm]"))
            print(f"📌 {os.path.basename(file_path)} → X={pos_x}, Y={pos_y}")
            elapsed_time = file_times[idx] - start_time

            process_single_file(
                file_path,
                ax,
                canvas,
                elapsed_time=elapsed_time,
                potential_range=params["plage_potentiel"],
                peak_range=params["position_pic"],
                lam=params["lambda"],
                method=params.get("baseline_method", "ARPLS"),
                polyorder=params.get("polyorder", 1),
                apply_baseline=True,
                window_length=params.get("window_length", 11),
                pos_x=pos_x,
                pos_y=pos_y
            )
        else:
            print("⚠️ Trop de fichiers SWV par rapport aux positions disponibles.")



def save_results_to_excel():

    if not results:
        print("📭 Aucun résultat à sauvegarder.")
        return

    # 1) DataFrame résultats
    df_results = pd.DataFrame(results)

    # 2) Préparer une clé de jointure "file_base" (basename normalisé)
    if "Nom du fichier" not in df_results.columns:
        print("⚠️ 'Nom du fichier' absent des résultats — sauvegarde interrompue.")
        return

    df_results["__file_base"] = df_results["Nom du fichier"].astype(str).str.strip().apply(lambda x: os.path.basename(x).lower())

    # 3) Récupérer DataFrame positions si disponible (priorité à shared_state)
    df_pos = None
    if shared_state.get("df_secm_positions") is not None:
        df_pos = shared_state["df_secm_positions"].copy()
    elif secm_positions:
        # si seulement la liste est présente, la transformer en DataFrame
        df_pos = pd.DataFrame(secm_positions)

    if df_pos is not None:
        # Normaliser noms colonnes possibles -> X_rel, Y_rel, File
        cols = {c: c for c in df_pos.columns}
        # méthode tolérante pour identifier colonnes X/Y
        for c in df_pos.columns:
            low = c.lower()
            if "x" in low and "rel" in low:
                cols[c] = "X_rel"
            if "y" in low and "rel" in low:
                cols[c] = "Y_rel"
            if "file" in low:
                cols[c] = "File"
        df_pos = df_pos.rename(columns=cols)

        # S'assurer qu'on a bien X_rel / Y_rel / File
        if "File" not in df_pos.columns:
            # tenter la dernière colonne comme fichier (comme dans ton code original)
            df_pos = df_pos.rename(columns={df_pos.columns[-1]: "File"})

        # normaliser File -> basename lower
        df_pos["File"] = df_pos["File"].astype(str).str.strip().apply(lambda x: os.path.basename(x).lower())

        # garder uniquement colonnes utiles
        df_pos = df_pos[["File", "X_rel", "Y_rel"]].copy()
        # renommer en colonnes destinées dans l'Excel (si tu veux l'affichage exact)
        df_pos = df_pos.rename(columns={"X_rel": "X_pos_merge", "Y_rel": "Y_pos_merge"})

        # créer clé de jointure
        df_pos["__file_base"] = df_pos["File"]
    else:
        df_pos = None

    # 4) Fusionner (merge) si on a les positions
    if df_pos is not None:

    if "Y rel. (µm)" not in df_final.columns:

                        worksheet.write_formula(excel_row - 1, col_idx, formula, format_courant)

            param_sheet.write('B10', K)
            param_sheet.write('A11', 'n')
            param_sheet.write('B11', n)

            param_sheet.write('A13', 'Formule courant normalisé:')
            param_sheet.write('B13', '= Hauteur_pic / Hauteur_pic_ref')
            param_sheet.write('A14', 'Formule gain (%):')
            param_sheet.write('B14', '= (Courant_normalisé - 1) * 100')

def recalculate_normalization():
    """
    Recalcule la normalisation pour TOUS les résultats en mémoire UNE SEULE FOIS.
    Appelée uniquement quand la référence ou l'index change.
    """
    if not results:
        return

    ref_h_external = shared_state.get("norm_ref_height")
    norm_index = shared_state.get("last_parameters", {}).get("normalisation_index", 4)

    # Déterminer la référence à utiliser
    if ref_h_external and ref_h_external > 0:
        # Priorité à la référence externe
        ref_value = float(ref_h_external)
        #print(f"🔢 Normalisation appliquée (ref externe) = {ref_value:.6g} µA pour {len(results)} résultats")
    elif 0 <= norm_index < len(results):
        # Fallback : index choisi dans l'UI
        ref_value = results[norm_index]["Peak height (µA)"]
        if not ref_value or ref_value == 0:
            print(f"⚠️ Index {norm_index} non valide (None ou 0). Impossible to apply normalization")
            return
        #print(f"🔢 Normalisation appliquée (index {norm_index}) = {ref_value:.6g} µA pour {len(results)} résultats")
    else:
        #print(f"⚠️ Index of normalization ({norm_index}) beyond limits (n={len(results)}). Normalization impossible.")
        return

    # Recalculer TOUS les résultats une seule fois
    for res in results:
        h = res.get("Peak height (µA)")
        if h is not None and ref_value and ref_value > 0:
            res["Courant normalisé"] = h / ref_value
            res["Signal change (%)"] = (res["Courant normalisé"] - 1.0) * 100.0

# --- Fonction pour tracer brut (non baseline, sans nettoyage outliers) ---
def plot_raw_selected_files():
    """
    Sélectionne des fichiers et trace le voltammogramme "brut" au sens:
    - sans nettoyage des outliers (pas de Hampel/MAD)
    - avec filtre Savitzky–Golay (UI)
    - avec auto-inversion (même logique que le traitement)
    - sans baseline ni corrections
    """
    ax = shared_state.get("ax")
    canvas = shared_state.get("canvas")
    if ax is None or canvas is None:
        print("⚠️ UI non initialisée.")
        return

    file_paths = filedialog.askopenfilenames(filetypes=[("Text files", "*.txt")])
    if not file_paths:
        print("❌ Aucun fichier sélectionné (brut).")
        return

    params = get_current_parameters()
    if params is None:
        print("⚠️ Paramètres invalides (brut).")
        return

    pot_min, pot_max     = params["plage_potentiel"]
    peak_range           = params["position_pic"]
    window_length        = params["window_length"]
    polyorder            = params["polyorder"]


    for file_path in file_paths:
        try:
            df = lire_donnees(file_path)
            y_series = (df['Courant (uA)'] if 'Courant (uA)' in df.columns else df['Courant inverse (uA)']).astype(float)
            y_raw = y_series.to_numpy(dtype=float)
            pot   = df['Potentiel (V)'].to_numpy(dtype=float)

            # Auto-inversion: même logique que le traitement
            mask_pot  = (pot >= pot_min) & (pot <= pot_max)
            mask_peak = (pot >= peak_range[0]) & (pot <= peak_range[1])
            mask_both = mask_pot & mask_peak

            try:
                if mask_both.any():
                    y_window = y_raw[mask_both]
                elif mask_pot.any():
                    y_window = y_raw[mask_pot]
                else:
                    y_window = y_raw

                invert = False
                if y_window.size > 0:
                    v = y_window[np.argmax(np.abs(y_window))]
                    invert = (v < 0)
            except Exception as e:
                print(f"⚠️ Auto-inversion ignorée (raison: {e})")
                invert = False

            # Mode superposition -> on masque le texte fixe du pic (sinon ça peut induire en erreur)
            ax = shared_state.get("ax")
            canvas = shared_state.get("canvas")
            update_fixed_peak_display(ax, canvas, visible=False)



            # Appliquer inversion et Savitzky définitif
            y_used     = (-y_raw if invert else y_raw)
            y_filtered = apply_savgol_filter(y_used, window_length=window_length, polyorder=polyorder)

            # Masque de tracé sur la plage UI
            mask = (pot >= pot_min) & (pot <= pot_max)
            if mask.any():
                pot_plot = pot[mask]
                y_plot   = y_filtered[mask]
            else:
                pot_plot = pot
                y_plot   = y_filtered

            ax.plot(pot_plot, y_plot,
                    label=f"Raw trace (SG, without anti-spikes): {os.path.basename(file_path)}",
                    alpha=0.85, linewidth=1.4)
            print(f"📈 Raw trace : {os.path.basename(file_path)}")
        except Exception as e:
            print(f"⚠️ Impossible de tracer brut pour {os.path.basename(file_path)} : {e}")

    ax.set_xlabel("Potential (V)")
    ax.set_ylabel("Current (µA)")
    ax.set_title("Voltammograms without Hampel outliers cleaning")
    ax.legend(loc="best", fontsize=8)
    canvas.draw_idle()

def update_gain_plot(ax_gain, canvas_gain, results):
    """
    Met à jour le graphique du gain en fonction du temps pour tous les fichiers traités.
    """
    if len(results) < 5:
        return  # Attendre un minimum de 5 fichiers avant de tracer

    # Recalculer les données pour le graphique
    times = [res["Cumulated time (s)"] for res in results]
    gains = [res["Signal change (%)"] for res in results]

    # Nettoyer l'axe précédent
    ax_gain.clear()
    ax_gain.plot(times, gains, marker='o', linestyle='-', color='blue', label="Signal change (%)")
    ax_gain.set_title("Signal Change (%) vs Time")
    ax_gain.set_xlabel("Cumulated time (s)")
    ax_gain.set_ylabel("Signal change (%)")
    ax_gain.legend()

    # --- CADRAGE ROBUSTE ---
    ax_gain.relim()

    # Ajouter une petite marge (évite "collé au bord" / hors cadre)
    ax_gain.margins(x=0.03, y=0.12)

    # Redessiner le graphique
    canvas_gain.draw_idle()


def generate_secm_map_dat(normalize=False, concentration_map=False, log_scale=False):
    """
    Génère une carte SECM en utilisant les positions chargées depuis le fichier .dat.

    - normalize=True   -> carte de Signal change (%) (sinon Hauteur brute)
    - concentration_map=True -> carte de concentration (Langmuir-Hill)
        * si log_scale=True  -> LogNorm sur la color map (ticks 10^… M), valeur au curseur en M
        * si log_scale=False -> échelle linéaire avec unité auto (nM/µM/mM/M), valeur au curseur dans la même unité

    Ajustement demandé:
    - Gains négatifs => concentration plancher de 1 nM (1e-9 M) pour le tracé.
    - En log: on borne aussi l’affichage à ≥ 1e-9 M pour éviter une échelle trop basse.
    """
    if not results:
        print("📭 Aucun résultat disponible pour générer la carte SECM .dat.")
        return
    if not secm_positions:
        print("⚠️ Aucune position SECM .dat chargée. Utilisez 'Charger positions SECM' d'abord.")
        return

    import matplotlib.pyplot as plt
    from scipy.interpolate import griddata
    from scipy.ndimage import gaussian_filter
    from matplotlib import colors as mcolors, ticker as mticker

    MIN_CONC_FLOOR = 1e-9  # 1 nM

    # Association fichier → position
    pos_map = shared_state.get("pos_by_file", {})

    # Filtrer uniquement les résultats avec positions valides
    valid_results = []
    for res in results:
        file_base = os.path.basename(res["Nom du fichier"]).lower()
        if file_base in pos_map or (res.get("X rel. (µm)") is not None and res.get("Y rel. (µm)") is not None):
            valid_results.append(res)
    if not valid_results:
        print("⚠️ Aucune correspondance trouvée entre résultats et positions .dat.")
        return

    xs = np.array([res["X rel. (µm)"] for res in valid_results], dtype=float)
    ys = np.array([res["Y rel. (µm)"] for res in valid_results], dtype=float)

    # Base: hauteur de pic
    heights = np.array([res["Peak height (µA)"] for res in valid_results], dtype=float)

    # === Mode Hauteur brute / Gain ===
    if not concentration_map:
        ref_h = shared_state.get("norm_ref_height")
        if normalize and len(heights) > 0:
            if ref_h and ref_h > 0:
                heights = ((heights / ref_h) - 1) * 100.0
                label_cbar = "Signal change (%)"
                print(f"🔹 Carte de gain basée sur référence externe: {ref_h:.3f} µA")
            else:
                ref_index = shared_state.get("last_parameters", {}).get("normalisation_index", 3)
                if 0 <= ref_index < len(valid_results):
                    ref_value = valid_results[ref_index]["Peak height (µA)"]
                    ref_x = valid_results[ref_index]["X rel. (µm)"]
                    ref_y = valid_results[ref_index]["Y rel. (µm)"]
                    print(f"🔹 Carte de gain basée sur le point {ref_index} ({ref_x:.0f} µm, {ref_y:.0f} µm) | Ref = {ref_value:.3f}")
                else:
                    ref_value = np.nanmax(heights)
                    print(f"⚠️ Index de normalisation invalide ({ref_index}), utilisation du maximum ({ref_value:.3f}) comme référence.")
                if ref_value and ref_value > 0:
                    heights = ((heights / ref_value) - 1) * 100.0
                label_cbar = "Signal change (%)"
        else:
            label_cbar = "Peak height (µA)"



    # === Mode Concentration (Langmuir‑Hill) ===
    label_cbar = "Concentration (M)" if concentration_map else label_cbar
    concentration = None  # en M
    if concentration_map:
        try:
            Vmax = float(shared_state["Vmax_var"].get())
            K    = float(shared_state["K_var"].get())
            n    = float(shared_state["n_var"].get())
        except Exception as e:
            print(f"⚠️ Erreur lecture Vmax/K/n : {e}")
            return

        # Gain (%) source
        ref_h = shared_state.get("norm_ref_height")
        if ref_h and ref_h > 0:
            # Gain basé sur la référence externe (prioritaire)
            gain_array = ((heights / float(ref_h)) - 1.0) * 100.0
            print(f"🔹 Carte de concentration basée sur référence externe: {float(ref_h):.3f} µA")
        else:
            # Fallback: index de normalisation courant (UI), borné
            ref_index = shared_state.get("last_parameters", {}).get("normalisation_index", 3)
            if 0 <= ref_index < len(valid_results):
                ref_value = valid_results[ref_index]["Peak height (µA)"]
                print(f"🔹 Carte de concentration basée sur le point {ref_index} (réf = {ref_value:.3f} µA)")
            else:
                ref_value = float(np.nanmax(heights))
                print(
                    f"⚠️ Index de normalisation invalide ({ref_index}), utilisation du maximum ({ref_value:.3f} µA) comme référence.")
            if ref_value and ref_value > 0:
                gain_array = ((heights / ref_value) - 1.0) * 100.0
            else:
                gain_array = np.full_like(heights, np.nan, dtype=float)

        # Politique:
        # - g < 0    -> C = 1 nM
        # - g = 0    -> C = 0 M (physique)
        # - 0 < g < g_upper -> conversion inverse
        # - g >= g_upper    -> saturer pour rester fini
        safety_margin = 0.01 * Vmax
        g_upper = Vmax - safety_margin

        concentration = np.full_like(gain_array, np.nan, dtype=float)

        mask_neg  = np.isfinite(gain_array) & (gain_array < 0)
        mask_zero = np.isfinite(gain_array) & (gain_array == 0)
        mask_mid  = np.isfinite(gain_array) & (gain_array > 0) & (gain_array < g_upper)
        mask_hi   = np.isfinite(gain_array) & (gain_array >= g_upper)

        concentration[mask_neg]  = MIN_CONC_FLOOR
        concentration[mask_zero] = 0.0

        with np.errstate(divide='ignore', invalid='ignore'):
            if np.any(mask_mid):
                ratio_mid = gain_array[mask_mid] / (Vmax - gain_array[mask_mid])
                concentration[mask_mid] = K * (ratio_mid ** (1.0 / n))
            if np.any(mask_hi):
                ratio_hi = g_upper / (Vmax - g_upper)
                concentration[mask_hi] = K * (ratio_hi ** (1.0 / n))

        # Nettoyage sécurité
        concentration[(~np.isfinite(concentration)) | (concentration < 0)] = 0.0
        # Sauvegarde
        for i, res in enumerate(valid_results):
            res["Concentration_calc (M)"] = concentration[i]

        heights = concentration.copy()  # base pour interpolation

    # Limites XY
    x_min, x_max = np.nanmin(xs), np.nanmax(xs)
    y_min, y_max = np.nanmin(ys), np.nanmax(ys)

    # Grille d'interpolation
    grid_size = 300
    xi = np.linspace(x_min, x_max, grid_size)
    yi = np.linspace(y_min, y_max, grid_size)
    xi_grid, yi_grid = np.meshgrid(xi, yi)
    # Limites arrondies à la centaine (définir AVANT les edges)
    x0 = float(np.floor(x_min / 100.0) * 100.0)
    x1 = float(np.ceil(x_max / 100.0) * 100.0)
    y0 = float(np.floor(y_min / 100.0) * 100.0)
    y1 = float(np.ceil(y_max / 100.0) * 100.0)

    # Construire des vecteurs d'arêtes (edges) couvrant exactement [x0, x1] et [y0, y1]
    dx = float(xi[1] - xi[0]) if xi.size > 1 else 1.0
    dy = float(yi[1] - yi[0]) if yi.size > 1 else 1.0

    x_edges = np.empty(xi.size + 1, dtype=float)
    y_edges = np.empty(yi.size + 1, dtype=float)

    x_edges[1:-1] = 0.5 * (xi[:-1] + xi[1:])
    x_edges[0] = xi[0] - dx / 2.0
    x_edges[-1] = xi[-1] + dx / 2.0

    y_edges[1:-1] = 0.5 * (yi[:-1] + yi[1:])
    y_edges[0] = yi[0] - dy / 2.0
    y_edges[-1] = yi[-1] + dy / 2.0

    # Forcer les bords sur tes limites arrondies
    x_edges[0], x_edges[-1] = x0, x1
    y_edges[0], y_edges[-1] = y0, y1

    # Interpolation (cascade)
    zi = None
    for method in ('cubic', 'linear', 'nearest'):
        try:
            zi_try = griddata((xs, ys), heights, (xi_grid, yi_grid), method=method)
            if zi is None and zi_try is not None:
                zi = zi_try
            if zi_try is not None and not np.all(np.isnan(zi_try)):
                zi = zi_try
                break
        except Exception as e:
            print(f"⚠️ Erreur interpolation ({method}): {e}")

    if zi is None:
        print("⚠️ Échec de l'interpolation.")
        return

    # Inpainting nearest des trous (hors hull, etc.)
    try:
        h, w = zi.shape
        J, I = np.mgrid[0:h, 0:w]
        valid = np.isfinite(zi)
        if not np.all(valid):
            points_valid = np.column_stack((I[valid], J[valid]))
            values_valid = zi[valid]
            points_all   = np.column_stack((I.ravel(), J.ravel()))
            zi_nearest   = griddata(points_valid, values_valid, points_all, method='nearest').reshape(h, w)
            inpaint_mask = ~valid
            zi = np.where(valid, zi, zi_nearest)
        else:
            inpaint_mask = np.zeros_like(zi, dtype=bool)
    except Exception as e:
        print(f"ℹ️ Inpainting nearest indisponible: {e}")
        inpaint_mask = np.zeros_like(zi, dtype=bool)

    # Clippage basique et lissage
    zi = np.where(np.isfinite(zi), zi, 0.0)
    zi[zi < 0] = 0.0
    zi_smooth = gaussian_filter(zi, sigma=3)


    # === Tracé (UNE seule colorbar) ===
    fig, ax = plt.subplots(figsize=SECM_FIGSIZE)
    shared_state["last_dat_map_fig"] = fig
    # Mémoriser la figure pour export vectoriel via l’UI
    shared_state["last_dat_map_fig"] = fig
    cmap = plt.get_cmap('seismic')


    # Utilitaire curseur
    def _nearest_val(x, y, xi_arr, yi_arr, z_grid):
        try:
            if not (np.isfinite(x) and np.isfinite(y)):
                return np.nan, (None, None)
            i = int(np.nanargmin(np.abs(xi_arr - x)))
            j = int(np.nanargmin(np.abs(yi_arr - y)))
            val = z_grid[j, i]
            return float(val) if np.isfinite(val) else np.nan, (j, i)
        except Exception:
            return np.nan, (None, None)

    show_hover = shared_state.get("show_hover_value_var").get() if shared_state.get("show_hover_value_var") else True

    if concentration_map:
        # Cmax à partir des points (non lissés)
        finite_conc = concentration[np.isfinite(concentration)] if concentration is not None else np.array([])
        Cmax_M = float(np.nanmax(finite_conc)) if finite_conc.size else 0.0

        # Lecture option échelle fixe (en M) depuis l’UI
        fixed = bool(shared_state.get("fixed_conc_enable_var") and shared_state["fixed_conc_enable_var"].get())
        vmin_user = float(shared_state["fixed_conc_min_var"].get()) if fixed else None
        vmax_user = float(shared_state["fixed_conc_max_var"].get()) if fixed else None

        # Validation simple
        if fixed and (not np.isfinite(vmin_user) or not np.isfinite(vmax_user) or vmin_user >= vmax_user):
            print("⚠️ Échelle de concentration fixe invalide (min/max). Option ignorée.")
            fixed = False
        if log_scale:
            # LogNorm: on borne l’affichage à ≥ 1 nM pour éviter des échelles 10^-18
            zi_plot = zi_smooth.copy()
            zi_plot = np.where(zi_plot > 0, zi_plot, MIN_CONC_FLOOR)

            # Déterminer vmin/vmax propres
            pos_vals = zi_plot[zi_plot > 0]
            zmin_pos = float(np.nanmin(pos_vals)) if pos_vals.size else MIN_CONC_FLOOR
            zmax_pos = float(np.nanmax(pos_vals)) if pos_vals.size else MIN_CONC_FLOOR
            zmin_pos = max(zmin_pos, MIN_CONC_FLOOR)

            vmin = max(MIN_CONC_FLOOR, 10.0 ** np.floor(np.log10(zmin_pos)))
            vmax = 10.0 ** np.ceil(np.log10(zmax_pos))
            if vmin >= vmax:
                vmax = vmin * 10.0

            norm = mcolors.LogNorm(vmin=vmin, vmax=vmax, clip=True)
            mesh = ax.pcolormesh(xi, yi, zi_plot, cmap=cmap, norm=norm, shading='auto')
            cbar = plt.colorbar(mesh, ax=ax, pad=0.08, shrink=1.0)
            cbar.set_label("Concentration (M)", fontsize=24, fontweight="bold")
            cbar.locator   = mticker.LogLocator(base=10, numticks=5)
            cbar.formatter = mticker.LogFormatterMathtext(base=10)
            cbar.update_ticks()


            if show_hover:
                def _fmt(x, y):
                    val, (j, i) = _nearest_val(x, y, xi, yi, zi_smooth)
                    if not np.isfinite(val) or val <= 0:
                        tag = " (estimé)" if (j is not None and inpaint_mask[j, i]) else ""
                        # Affiche 0 là où la vraie concentration est 0 (g=0)
                        return f"x={x:.2f} µm, y={y:.2f} µm, C=0 M{tag}"
                    tag = " (estimé)" if (j is not None and inpaint_mask[j, i]) else ""
                    return f"x={x:.2f} µm, y={y:.2f} µm, C={val:.3e} M{tag}"
                ax.format_coord = _fmt


        else:
            # Linéaire avec unité auto
            finite_vals = zi_smooth[np.isfinite(zi_smooth)]
            if finite_vals.size == 0:
                print("⚠️ Pas de valeurs finies pour l'échelle linéaire.")
                return
            max_conc = float(np.nanmax(finite_vals))
            if max_conc < 1e-6 and max_conc > 0:
                scale = 1e9; unit_txt = "nM"
            elif max_conc < 1e-3 and max_conc > 0:
                scale = 1e6; unit_txt = "µM"
            elif max_conc < 1 and max_conc > 0:
                scale = 1e3; unit_txt = "mM"
            else:
                scale = 1.0; unit_txt = "M"

            zi_lin = zi_smooth * scale

            if fixed:
                # Impose vmin/vmax dans l’unité affichée (nM/µM/mM/M)
                vmin_lin = vmin_user * scale
                vmax_lin = vmax_user * scale
                norm = mcolors.Normalize(vmin=vmin_lin, vmax=vmax_lin, clip=True)
                mesh = ax.pcolormesh(x_edges, y_edges, zi_lin, cmap=cmap, norm=norm, shading='flat')
            else:
                mesh = ax.pcolormesh(x_edges, y_edges, zi_lin, cmap=cmap, shading='flat')

            cbar = plt.colorbar(mesh, ax=ax, pad=0.08, shrink=1.0)
            cbar.set_label(f"Concentration ({unit_txt})", fontsize=24, fontweight="bold")

            # Ticks (3): si échelle fixe → basés sur vmin/vmax fixés ; sinon → basés sur les données
            if fixed:
                t0 = 0.0 if np.isclose(vmin_lin, 0.0) else vmin_lin
                t2 = 100.0 * np.floor(vmax_lin / 100.0)
                t2 = min(t2, vmax_lin)
                # Milieu exact (pour 0–500 => 250), sans arrondi pour garder l’équidistance visuelle
                t1 = 0.5 * (t0 + t2)
                ticks = [t0, t1, t2]
                cbar.set_ticks(ticks)
                cbar.set_ticklabels([f"{int(round(t)):d}" if t >= 100 else f"{t:.0f}" for t in ticks])
            else:
                vmin_lin = float(np.nanmin(zi_lin))
                vmax_lin = float(np.nanmax(zi_lin))
                if not np.isfinite(vmin_lin) or not np.isfinite(vmax_lin) or vmin_lin == vmax_lin:
                    vmin_lin, vmax_lin = 0.0, 1.0
                t0 = 0.0
                t2 = 100.0 * np.floor(vmax_lin / 100.0)
                if t2 <= t0:
                    t2 = vmax_lin
                t1 = (t0 + t2) / 2.0
                if t2 >= 200.0:
                    t1 = 100.0 * np.round(t1 / 100.0)
                t2 = min(t2, vmax_lin)
                t1 = min(max(t1, t0), t2)
                ticks = [t0, t1, t2]
                cbar.set_ticks(ticks)
                cbar.set_ticklabels([f"{int(round(t)):d}" if t >= 100 else f"{t:.0f}" for t in ticks])


            for lbl in cbar.ax.get_yticklabels():
                lbl.set_fontsize(20)
                lbl.set_fontweight("bold")

            # Afficher C_max au-dessus de la colorbar (hors carte), dans l’unité courante
            if np.isfinite(Cmax_M) and Cmax_M >= 0:
                if scale != 1.0:
                    cbar.ax.set_title(f"C_max = {Cmax_M * scale:.3g} {unit_txt}",
                                      fontsize=12, fontweight="bold", pad=20)
                else:
                    cbar.ax.set_title(f"C_max = {Cmax_M:.3e} M", fontsize=16, fontweight="bold", pad=20)

            # Affichage de la valeur au survol de la colorbar (linéaire, unité zi_lin => unit_txt: nM/µM/mM/M)
            if show_hover:
                def _fmt_cbar_lin(x, y, _unit=unit_txt):
                    if not np.isfinite(y):
                        return f"C=nan {_unit}"
                    return f"C={y:.3g} {_unit}"

                cbar.ax.format_coord = _fmt_cbar_lin

            if show_hover:
                def _fmt(x, y, _zi_lin=zi_lin, _xi=xi, _yi=yi, _unit=unit_txt):
                    val, (j, i) = _nearest_val(x, y, _xi, _yi, _zi_lin)
                    if not np.isfinite(val):
                        return f"x={x:.2f} µm, y={y:.2f} µm, C=nan"
                    tag = " (estimé)" if (j is not None and inpaint_mask[j, i]) else ""
                    # Si la concentration réelle est 0 (g=0), afficher 0
                    val_raw, _ = _nearest_val(x, y, _xi, _yi, zi_smooth)
                    if np.isfinite(val_raw) and val_raw == 0:
                        return f"x={x:.2f} µm, y={y:.2f} µm, C=0 {_unit}{tag}"
                    return f"x={x:.2f} µm, y={y:.2f} µm, C={val:.3g} {_unit}{tag}"
                ax.format_coord = _fmt


    else:
        # Hauteur brute / Gain
        mesh = ax.pcolormesh(x_edges, y_edges, zi_smooth, cmap=cmap, shading='flat')
        cbar = plt.colorbar(mesh, ax=ax, pad=0.08, shrink=1.0)
        cbar.set_label(label_cbar, fontsize=24, fontweight="bold")
        if "Gain" in label_cbar:
            # Choisis 4 ticks (change numticks à 3 si tu préfères 3 ticks)
            cbar.locator = mticker.LinearLocator(numticks=3)
            # Libellés arrondis à la dizaine, sans modifier l'échelle
            cbar.formatter = tens_formatter  # défini en global: FuncFormatter(lambda v, pos: f"{int(np.round(v/10)*10)}")
            cbar.update_ticks()
            # Ticks de la colorbar (Gain %) en gras et plus grands
            cbar.ax.tick_params(axis='y', labelsize=20)
            for lbl in cbar.ax.get_yticklabels():
                lbl.set_fontweight('bold')
            # Affichage de la valeur au survol de la colorbar (Gain %)
            if show_hover:
                def _fmt_cbar_gain(x, y):
                    # y est la valeur sur l’axe de la colorbar (en %)
                    if not np.isfinite(y):
                        return "Gain=nan %"
                    return f"Gain={y:.2f} %"
                cbar.ax.format_coord = _fmt_cbar_gain
        else:
            cbar.locator = mticker.LinearLocator(numticks=5)
            cbar.update_ticks()
            # Ticks de la colorbar (Hauteur du pic) en gras et plus grands
            cbar.ax.tick_params(axis='y', labelsize=20)
            for lbl in cbar.ax.get_yticklabels():
                lbl.set_fontweight('bold')

        if show_hover:
            if "Gain" in label_cbar:
                def _fmt(x, y):
                    val, _ = _nearest_val(x, y, xi, yi, zi_smooth)
                    if not np.isfinite(val):
                        return f"x={x:.2f} µm, y={y:.2f} µm, Gain=nan"
                    return f"x={x:.2f} µm, y={y:.2f} µm, Gain={val:.2f} %"
                ax.format_coord = _fmt
            else:
                def _fmt(x, y):
                    val, _ = _nearest_val(x, y, xi, yi, zi_smooth)
                    if not np.isfinite(val):
                        return f"x={x:.2f} µm, y={y:.2f} µm, I=nan"
                    return f"x={x:.2f} µm, y={y:.2f} µm, I={val:.3f} µA"
                ax.format_coord = _fmt

    ax.set_xlabel("X (µm)", fontsize=24, fontweight="bold")
    ax.set_ylabel("Y (µm)", fontsize=24, fontweight="bold")

    # Limites arrondies à la centaine (vers l'extérieur pour bien couvrir les données)
    x0 = np.floor(x_min / 100.0) * 100.0
    x1 = np.ceil(x_max / 100.0) * 100.0
    y0 = np.floor(y_min / 100.0) * 100.0
    y1 = np.ceil(y_max / 100.0) * 100.0

    ax.set_xlim(x0, x1)
    ax.set_ylim(y0, y1)
    ax.set_aspect('equal', adjustable='box')

    ax.xaxis.set_major_locator(LinearLocator(numticks=3))
    ax.yaxis.set_major_locator(LinearLocator(numticks=3))
    ax.xaxis.set_major_formatter(FormatStrFormatter('%.0f'))
    ax.yaxis.set_major_formatter(FormatStrFormatter('%.0f'))
    # Style des ticks des axes X/Y
    ax.tick_params(axis="both", labelsize=20)
    for lbl in ax.get_xticklabels():
        lbl.set_fontweight("bold")
    for lbl in ax.get_yticklabels():
        lbl.set_fontweight("bold")
    plt.tight_layout()
    plt.show()

def open_gain_window():
    global gain_window, ax_gain, canvas_gain

    if gain_window is not None:
        try:
            if hasattr(gain_window, 'after_id') and gain_window.after_id is not None:
                gain_window.after_cancel(gain_window.after_id)
            gain_window.destroy()
        except Exception:
            pass
        gain_window = None

    gain_window = tk.Toplevel()  # Utiliser Toplevel pour une vraie pop-up
    gain_window.title("Évolution du Gain")
    gain_window.geometry(GAIN_WINDOW_SIZE)

    # Frame pour graphique et boutons
    frame_main = ttk.Frame(gain_window, padding=10)
    frame_main.pack(fill="both", expand=True)

    # Graphique
    fig_gain = Figure(figsize=GAIN_FIGSIZE)
    ax_gain = fig_gain.add_subplot(111)
    canvas_gain = FigureCanvasTkAgg(fig_gain, master=frame_main)
    canvas_widget_gain = canvas_gain.get_tk_widget()
    canvas_widget_gain.pack(fill="both", expand=True)

    # --- Toolbar Matplotlib (comme carte SECM) ---
    toolbar = NavigationToolbar2Tk(canvas_gain, frame_main)
    toolbar.update()
    toolbar.pack(side="bottom", fill="x")

    gain_window.after_id = None

    def refresh_plot():
        if len(results) >= 5:
            update_gain_plot(ax_gain, canvas_gain, results)
        if gain_window.winfo_exists():
            gain_window.after_id = gain_window.after(2000, refresh_plot)

    def on_close():
        if gain_window.after_id is not None:
            try:
                gain_window.after_cancel(gain_window.after_id)
            except Exception:
                pass
        gain_window.destroy()
        shared_state["gain_window_open"] = False

    gain_window.protocol("WM_DELETE_WINDOW", on_close)
    refresh_plot()

def reset_application(ax, canvas):
    global results, shared_state, gain_window, root
    """
    Réinitialise l'application en vidant les résultats, réinitialisant les graphiques,
    et permettant à l'utilisateur de redémarrer.
    """
    if len(results) > 20:
        from tkinter import messagebox
        if root and root.winfo_exists():
            response = messagebox.askyesno(
                "⚠️ Confirmer la réinitialisation",
                f"Vous avez {len(results)} résultats en mémoire.\n\n"
                "Êtes-vous sûr de vouloir réinitialiser ?\n"
                "(Les résultats non exportés seront perdus)"
            )
            if not response:  # Utilisateur a cliqué "Non"
                print("❌ Réinitialisation annulée par l'utilisateur.")
                return  # ← Sort de la fonction, ne réinitialise pas

    # Réinitialiser les résultats
    results.clear()
    shared_state["stop"] = True  # Arrêter toute surveillance en cours
    shared_state["gain_window_open"] = False  # Réinitialiser l'état de la fenêtre du gain
    shared_state["directory"] = None

    # Fermer la fenêtre du gain de manière sécurisée
    if gain_window:
        try:
            # Annule le after si programmé
            if hasattr(gain_window, 'after_id') and gain_window.after_id is not None:
                try:
                    gain_window.after_cancel(gain_window.after_id)
                except Exception:
                    pass
            # Détruit la fenêtre si elle existe encore (ignore l’erreur si déjà détruite)
            try:
                if root is not None and hasattr(gain_window, "destroy"):
                    root.after(0, gain_window.destroy)
            except Exception as e:
                if "application has been destroyed" not in str(e):
                    print(f"⚠️ Erreur lors de la destruction de la fenêtre du gain : {e}")
        except Exception as e:
